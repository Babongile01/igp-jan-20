name of the job --> ci-job

ci script --> 

pipeline
{
	agent any
	stages
	{
		stage('Checkout')
		{
			steps
			{
				git  'https://github.com/jsachdev07/igpaug23.git'
			}
		}
		
		stage('Compile')
		{
			steps
			{
				sh 'mvn compile'
			}
		}

		stage('Test')
		{
			steps
			{
				sh 'mvn test'
			}
		}

		stage('Packaging')
		{
			steps
			{
				sh 'mvn package'
			}
		}
     }
}
